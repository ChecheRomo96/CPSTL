141 // Largest value of an unsigned type and of a signed type with the
142 // same width, computed without overflow.
143 template<class T>
144 constexpr T unsigned_max() noexcept { return static_cast<T>(~static_cast<T>(0)); }

146 template<class T>
147 constexpr T signed_max() noexcept {
148 return static_cast<T>(((static_cast<T>(1) << (sizeof(T) * CHAR_BIT - 2)) - 1) * 2 + 1);
